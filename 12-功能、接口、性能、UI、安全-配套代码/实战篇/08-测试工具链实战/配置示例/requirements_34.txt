pipeline {
    agent any
    stages {
        stage('Install Dependencies') {
            steps {
                sh 'pip install -r requirements.txt'
            }
        }
        stage('Initialize Database') {
            steps {
                sh 'mysql -h 127.0.0.1 -u root -p123456 seckill_finance_db < schema.sql'
            }
        }
        stage('Run Tests') {
            steps {
                sh 'python test_main_flow.py'
            }
        }
    }
    post {
        always {
            archiveArtifacts artifacts: 'reports/*.html'
        }
    }
}
