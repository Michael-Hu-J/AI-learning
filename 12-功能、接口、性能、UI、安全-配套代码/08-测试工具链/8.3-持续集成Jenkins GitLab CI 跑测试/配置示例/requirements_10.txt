import requests
import json

def send_feishu_notification(title, content, webhook_url):
    """发送飞书通知"""
    data = {
        "msg_type": "interactive",
        "card": {
            "header": {
                "title": {"tag": "plain_text", "content": title},
                "template": "red"  # 红色标题（失败）/ green（成功）
            },
            "elements": [
                {"tag": "div", "text": {"tag": "lark_md", "content": content}}
            ]
        }
    }
    requests.post(webhook_url, json=data)

# 测试失败时调用
